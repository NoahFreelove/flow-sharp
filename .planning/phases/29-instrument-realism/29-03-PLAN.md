    using System;
    using System.Numerics;
    using FlowLang.StandardLibrary.Audio;

    namespace FlowLang.Tests.Helpers;

    /// <summary>
    /// Phase 29 — hand-rolled radix-2 FFT + spectral helpers for the velocity-layer
    /// and harmonic-richness tests. No new external dependency (per SPEC D-32).
    /// </summary>
    public static class Phase29Fft
    {
        /// <summary>
        /// Recursive radix-2 Cooley-Tukey FFT. Input length MUST be a power of 2.
        /// For 1-sec / 44.1 kHz buffers, pad to 65536 samples (next power of 2 above 44100).
        /// </summary>
        public static Complex[] Fft(Complex[] x)
        {
            int n = x.Length;
            if ((n & (n - 1)) != 0) throw new ArgumentException("Length must be power of 2");
            if (n == 1) return x;

            var even = new Complex[n / 2];
            var odd = new Complex[n / 2];
            for (int i = 0; i < n / 2; i++)
            {
                even[i] = x[2 * i];
                odd[i] = x[2 * i + 1];
            }
            var evenT = Fft(even);
            var oddT = Fft(odd);

            var output = new Complex[n];
            for (int k = 0; k < n / 2; k++)
            {
                var twiddle = Complex.FromPolarCoordinates(1.0, -2.0 * Math.PI * k / n) * oddT[k];
                output[k] = evenT[k] + twiddle;
                output[k + n / 2] = evenT[k] - twiddle;
            }
            return output;
        }

        /// <summary>
        /// Compute magnitude spectrum of an AudioBuffer. Pads to next power of 2.
        /// Returns magnitudes for bins 0..N/2 (real-input symmetry).
        /// </summary>
        public static double[] ComputeMagnitudeSpectrum(AudioBuffer buffer)
        {
            int n = NextPowerOf2(buffer.Frames);
            var input = new Complex[n];
            // AudioBuffer.Data is interleaved; for mono, channel 0 only
            // For stereo, mix to mono before FFT
            int channels = buffer.Channels;
            for (int i = 0; i < buffer.Frames; i++)
            {
                double sample = channels == 1
                    ? buffer.Data[i]
                    : (buffer.Data[i * 2] + buffer.Data[i * 2 + 1]) / 2.0;
                input[i] = new Complex(sample, 0);
            }
            var spectrum = Fft(input);
            var mag = new double[n / 2 + 1];
            for (int i = 0; i <= n / 2; i++) mag[i] = spectrum[i].Magnitude;
            return mag;
        }

        /// <summary>
        /// Cosine similarity between two magnitude spectra. Returns value in [-1, 1].
        /// For audio realism: closer to 1 = same timbre; closer to 0 = different timbre.
        /// </summary>
        public static double CosineSimilarity(double[] a, double[] b)
        {
            int n = Math.Min(a.Length, b.Length);
            double dot = 0, normA = 0, normB = 0;
            for (int i = 0; i < n; i++)
            {
                dot += a[i] * b[i];
                normA += a[i] * a[i];
                normB += b[i] * b[i];
            }
            if (normA == 0 || normB == 0) return 0;
            return dot / (Math.Sqrt(normA) * Math.Sqrt(normB));
        }

        /// <summary>
        /// Harmonic-richness ratio: sum of partial energies above fundamental / fundamental energy.
        /// </summary>
        public static double HarmonicRichnessRatio(AudioBuffer buffer, double fundamentalHz)
        {
            var mag = ComputeMagnitudeSpectrum(buffer);
            int n = NextPowerOf2(buffer.Frames);
            int fundBin = (int)Math.Round(fundamentalHz * n / buffer.SampleRate);
            if (fundBin < 1 || fundBin >= mag.Length) return 0;
            double fundEnergy = mag[fundBin] * mag[fundBin];
            if (fundEnergy < 1e-12) return 0;
            double harmEnergy = 0;
            for (int k = 2; fundBin * k < mag.Length; k++)
            {
                double m = mag[fundBin * k];
                harmEnergy += m * m;
            }
            return harmEnergy / fundEnergy;
        }

        private static int NextPowerOf2(int n)
        {
            int p = 1;
            while (p < n) p *= 2;
            return p;
        }
    }
