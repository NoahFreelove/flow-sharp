      Implementation:
      ```csharp
      var sigAbc = new FunctionSignature("abc", [StringType.Instance], ParameterNames: ["source"]);
      registry.Register("abc", sigAbc, args =>
      {
          if (!context.NotationIoEnabled)
              throw new InvalidOperationException("abc requires `use \"@notation-io\"`");
          var src = args[0].As<string>();
          // Count X: headers to dispatch single vs multi
          var xCount = Regex.Matches(src, @"^X:\s*\d+", RegexOptions.Multiline).Count;
          if (xCount >= 2)
          {
              var sections = AbcImport.ParseMultiTune(src);
              return Value.Array(SectionType.Instance, sections.Select(s => Value.Section(s)).ToList());
          }
          else
          {
              var section = AbcImport.ParseSingleTune(src);
              return Value.Section(section);
          }
      });
      ```

      The `flow-lang/notation-io.flow` forward decl for `abc` was already added in Plan 39-01 T6.
