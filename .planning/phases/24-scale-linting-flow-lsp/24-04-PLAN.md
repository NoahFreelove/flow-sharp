```csharp
public interface IDiagnosticsPublisher
{
    void Publish(DocumentUri uri, IReadOnlyList<FlowError> errors);
}

public sealed class DiagnosticsPublisher : IDiagnosticsPublisher
{
    private readonly ILanguageServerFacade _server;
    public DiagnosticsPublisher(ILanguageServerFacade server) => _server = server;

    public static IReadOnlyList<Diagnostic> BuildDiagnostics(IReadOnlyList<FlowError> errors)
    {
        var list = new List<Diagnostic>(errors.Count);
        foreach (var e in errors)
        {
            list.Add(new Diagnostic
            {
                Severity = LspMappings.ToSeverity(e.Level),
                Source = "flow",
                Message = e.Message,
                Range = LspMappings.ToRange(e.Location)
            });
        }
        return list;
    }

    public void Publish(DocumentUri uri, IReadOnlyList<FlowError> errors)
    {
        // MUST publish even when empty — that is how LSP clears prior markers.
        var diags = BuildDiagnostics(errors);
        _server.TextDocument.PublishDiagnostics(new PublishDiagnosticsParams
        {
            Uri = uri,
            Diagnostics = new Container<Diagnostic>(diags)
        });
    }
}
```

```csharp
using FlowLsp.Handlers;
using OmniSharp.Extensions.LanguageServer.Protocol;
using OmniSharp.Extensions.LanguageServer.Protocol.Document;
using OmniSharp.Extensions.LanguageServer.Protocol.Models;
using OmniSharp.Extensions.LanguageServer.Protocol.Server;

namespace FlowLsp.Diagnostics;

/// <summary>
/// Phase 24 Plan 24-04: orchestrator owning the single wire-level
/// _server.TextDocument.PublishDiagnostics call per URI per parse cycle.
///
/// LSP publishDiagnostics REPLACES per-URI (DiagnosticsPublisher.cs:52 comment).
/// If parse errors and scale-lint diagnostics each tried to publish independently,
/// the second call would clobber the first. This class composes both diagnostic
/// sources into one Container so editors see them simultaneously.
///
/// Source-tag separation is preserved: parse errors keep Source="flow"
/// (via DiagnosticsPublisher.BuildDiagnostics), scale-lint keeps
/// Source="flow.scaleLint" (via ScaleLintAnalyzer / IScaleLintPublisher).
/// Editors can filter independently.
///
/// Empty-publish-clears-squiggles invariant (DiagnosticsPublisher.cs:52) is
/// preserved by always pushing the merged list — empty-merged still publishes,
/// which is how LSP clears prior markers.
/// </summary>
public sealed class CombinedDiagnosticsPublisher
{
    private readonly ILanguageServerFacade _server;
    private readonly IScaleLintPublisher _lint;

    public CombinedDiagnosticsPublisher(ILanguageServerFacade server, IScaleLintPublisher lint)
    {
        _server = server;
        _lint = lint;
    }

    /// <summary>
    /// Pure: build the union of parse-error diagnostics and scale-lint diagnostics.
    /// Exposed static so unit tests can exercise the merge without standing up an
    /// ILanguageServerFacade — mirrors the static BuildDiagnostics on
    /// DiagnosticsPublisher. The static form bypasses dependency injection by
    /// invoking ScaleLintAnalyzer directly (Wave 0 ParseSession populates Pragmas).
    /// </summary>
    public static IReadOnlyList<Diagnostic> BuildAll(ParseResult result, string source)
    {
        var parseDiags = DiagnosticsPublisher.BuildDiagnostics(result.Errors);
        var lintDiags = ScaleLintAnalyzer.Analyze(result.Ast, result.Tokens, source);
        if (parseDiags.Count == 0 && lintDiags.Count == 0)
            return Array.Empty<Diagnostic>();
        var merged = new List<Diagnostic>(parseDiags.Count + lintDiags.Count);
        merged.AddRange(parseDiags);
        merged.AddRange(lintDiags);
        return merged;
    }

    /// <summary>
    /// Wire-level publish: build the union and call PublishDiagnostics ONCE.
    /// MUST publish even when the merged list is empty — that is how LSP clears
    /// prior markers (parse-error squiggles AND scale-lint squiggles).
    /// </summary>
    public void Publish(DocumentUri uri, ParseResult result, string source)
    {
        var parseDiags = DiagnosticsPublisher.BuildDiagnostics(result.Errors);
        var lintDiags = _lint.Analyze(result, source);
        var merged = new List<Diagnostic>(parseDiags.Count + lintDiags.Count);
        merged.AddRange(parseDiags);
        merged.AddRange(lintDiags);

        // MUST publish even when empty — that is how LSP clears prior markers.
        _server.TextDocument.PublishDiagnostics(new PublishDiagnosticsParams
        {
            Uri = uri,
            Diagnostics = new Container<Diagnostic>(merged)
        });
    }
}
```
