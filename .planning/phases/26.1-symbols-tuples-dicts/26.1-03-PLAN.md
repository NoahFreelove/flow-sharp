```csharp
using System.Linq;
using FlowLang.TypeSystem.PrimitiveTypes;

namespace FlowLang.TypeSystem.SpecialTypes;

/// <summary>
/// Tuple type with per-position element types and arity (Phase 26.1 TUP-09).
/// Empty `Tuple<<>>` and singleton `Tuple<<T>>` are valid arities.
/// IsHashable recurses to all elements — Tuple-of-hashables is a valid Dict key.
/// AnyArity sentinel matches any tuple via IsCompatibleWith (mirrors ArrayType(VoidType.Instance) precedent).
/// </summary>
public sealed class TupleType : FlowType
{
    public IReadOnlyList<FlowType> ElementTypes { get; }
    public bool IsAnyArity { get; }

    public TupleType(IReadOnlyList<FlowType> elementTypes)
    {
        ElementTypes = elementTypes ?? throw new ArgumentNullException(nameof(elementTypes));
        IsAnyArity = false;
    }

    private TupleType(IReadOnlyList<FlowType> elementTypes, bool isAnyArity)
    {
        ElementTypes = elementTypes;
        IsAnyArity = isAnyArity;
    }

    /// <summary>Sentinel matching any tuple type via IsCompatibleWith — used for `(unpack)` registration in Wave 3.</summary>
    public static TupleType AnyArity { get; } = new(System.Array.Empty<FlowType>(), isAnyArity: true);

    public override string Name => IsAnyArity
        ? "Tuple<<*>>"
        : $"Tuple<<{string.Join(", ", ElementTypes.Select(t => t.Name))}>>";

    public override bool Equals(FlowType? other)
    {
        if (other is not TupleType t) return false;
        if (IsAnyArity || t.IsAnyArity) return true;
        if (ElementTypes.Count != t.ElementTypes.Count) return false;
        for (int i = 0; i < ElementTypes.Count; i++)
            if (!ElementTypes[i].Equals(t.ElementTypes[i])) return false;
        return true;
    }

    public override int GetHashCode()
    {
        var hc = new HashCode();
        hc.Add(GetType());
        foreach (var et in ElementTypes) hc.Add(et);
        return hc.ToHashCode();
    }

    public override bool IsCompatibleWith(FlowType target)
    {
        if (target is not TupleType tt) return false;
        if (IsAnyArity || tt.IsAnyArity) return true;
        if (ElementTypes.Count != tt.ElementTypes.Count) return false;
        for (int i = 0; i < ElementTypes.Count; i++)
            if (!ElementTypes[i].IsCompatibleWith(tt.ElementTypes[i])) return false;
        return true;
    }

    public override bool CanConvertTo(FlowType target) => IsCompatibleWith(target);

    public override int GetSpecificity()
    {
        if (IsAnyArity) return 50;
        return ElementTypes.Sum(e => e.GetSpecificity()) + 60;
    }

    public override bool IsHashable() => !IsAnyArity && ElementTypes.All(e => e.IsHashable());
}
```

{
    if (c != '<' && c != '>') return null;
    if (PeekNext() != c) return null;
    // REVISION 1 (plan-checker BLOCKER fix): the gate must admit value-end tokens
    // that can legitimately precede a destructure statement on the next line:
    //   Int x = 5         <-- ends with IntLiteral
    //   <<Int a>> = ...   <-- destructure follows, must lex `<<`

    // and GreaterGreater (closing tuple from prev expr). This does NOT admit

    // path is governed separately at Parser.NoteStream.cs and the lexer-level
    // gate is moot. None of {NoteLiteral, Underscore, Plus, Minus} appear below.

**Note-stream protection (RESEARCH § Pitfall 1) — REVISION 1 update:** Inside note streams the previous emitted token is typically `NoteLiteral` (when the accent immediately follows a note like `C4q>`), `Underscore`, `Plus`/`Minus` (for `+5c`/`-5st` typed-literals).

REVISION-1 CAUTION: the extended gate above DOES admit `IntLiteral` and `NoteLiteral` as valid `<<` predecessors. This is required so the destructure-after-prev-statement case parses (`Int x = 5\n<<Int a>> = ...`). Inside note streams, the lexer's char-level path is the SAME shared path; whether we're "inside a note stream" is decided by Parser.NoteStream.cs at parse time, not by the lexer.

The note-stream `>` accent is one CHARACTER. PeekNext() inside `| C4q> D4q |` is the SPACE after `>` (or in `| C4q> |` it's the space before the closing pipe). NEITHER position has a second `>` after the first, so `TryLexAngleAngle` returns null at line 2 (`if (PeekNext() != c) return null;`) and the lexer falls through to single-char `>`. The gate is the second line of defense for the dual-`>>` case. CRITICAL CLARIFICATION: re-read RESEARCH § Q2. The claim is that type-annotation context never emits `LessLess`/`GreaterGreater` because the lexer's gate excludes the position. But `Identifier` IS in the gate's permitted set. So `Tuple<<` actually WILL emit `LessLess` in the lexer (Tuple Identifier → `<` after Identifier triggers the two-char check). To avoid this, the TypeParser must accept BOTH single-`<<` (LessLess token) and double-`< <` (two LessThan tokens) for backward compatibility with the lexer's gate behavior. The pragmatic path: support both forms.

{

    // Tuple delimiters can lex as LessLess (post-Identifier the gate allows two-char)
    // OR as two LessThan tokens (in nested contexts where the gate fires differently).
    // Support both forms for robustness.
    bool consumedLessLess = false;
    if (index < tokens.Count && tokens[index].Type == TokenType.LessLess)
    {
        index++; // consumed <<
        consumedLessLess = true;
    }
    else
    {
        if (index >= tokens.Count || tokens[index].Type != TokenType.LessThan)
            throw new ParseException($"Expected '<<' after Tuple at {tokens[index].Location}");
        index++;
        if (index >= tokens.Count || tokens[index].Type != TokenType.LessThan)
            throw new ParseException($"Expected '<<' after Tuple (second '<') at {tokens[index].Location}");
        index++;
    }

    var elementTypes = new List<FlowType>();
    bool first = true;
    // Stop when we see closing >> (either as GreaterGreater token or two GreaterThan)
    while (index < tokens.Count
           && tokens[index].Type != TokenType.GreaterThan
           && tokens[index].Type != TokenType.GreaterGreater)
    {
        if (!first)
        {
            if (tokens[index].Type != TokenType.Comma)
                throw new ParseException($"Expected ',' between Tuple element types at {tokens[index].Location}");
            index++;
        }
        var (elemType, next, _) = ParseType(tokens, index);
        elementTypes.Add(elemType);
        index = next;
        first = false;
    }

    if (index < tokens.Count && tokens[index].Type == TokenType.GreaterGreater)
    {
        index++; // single >> token
    }
    else
    {
        if (index >= tokens.Count || tokens[index].Type != TokenType.GreaterThan)
            throw new ParseException($"Expected '>>' after Tuple element types at {tokens[index].Location}");
        index++;
        if (index >= tokens.Count || tokens[index].Type != TokenType.GreaterThan)
            throw new ParseException($"Expected '>>' after Tuple element types (second '>') at {tokens[index].Location}");
        index++;

    var location = PreviousToken.Location;

    while (!Check(TokenType.GreaterGreater) && !IsAtEnd())

private void ExecuteTupleDestructure(TupleDestructureStatement stmt)
{
    var rhs = _evaluator.Evaluate(stmt.Value);
    if (rhs.Data is not IReadOnlyList<Value> tupArr || rhs.Type is not TupleType)
    {
        _errorReporter.ReportError(
            $"Right-hand side of destructure must be a Tuple, got {rhs.Type}",
            stmt.Location);
        return;
    }
    if (tupArr.Count != stmt.Patterns.Count)
    {
        _errorReporter.ReportError(
            $"Tuple destructure arity mismatch: pattern has {stmt.Patterns.Count} slot(s), value has {tupArr.Count}",
            stmt.Location);
        return;
    }
    for (int i = 0; i < stmt.Patterns.Count; i++)
    {
        var pattern = stmt.Patterns[i];
        var component = tupArr[i];
        if (pattern.Type != null
            && !component.Type.IsCompatibleWith(pattern.Type)
            && !component.Type.CanConvertTo(pattern.Type))
        {
            _errorReporter.ReportError(
                $"Cannot bind tuple component {i} of type {component.Type} to {pattern.Type} {pattern.Name}",
                stmt.Location);
            return;
        }
        if (pattern.Type != null
            && !component.Type.Equals(pattern.Type)
            && component.Type.CanConvertTo(pattern.Type))
        {
            component = component.ConvertTo(pattern.Type);
        }
        _context.DeclareVariable(pattern.Name, component);
    }
}
