```csharp
private static void RegisterReverb(InternalFunctionRegistry registry)
{
    // Existing reverbSimpleSig + reverbFullSig — UNCHANGED
    // ... existing 2 registrations ...

    // NEW Phase 26.2 ERG-02: reverb(Buffer, Double, Second) — decay time as Second
    // Per RESEARCH Pitfall 3 score analysis: this does NOT ambiguate with reverb(Buffer, Double, Double):
    //   - bare-Double call (reverb buf 0.5 1.5) wins exact-match Buffer/Double/Double = 3000 vs Buffer/Double/Second = 2500
    //   - Second-typed call (reverb buf 0.5 1.5s) wins exact-match Buffer/Double/Second = 3000 vs bare-Double = 2500
    // Use ReverbFull lambda but pass damping=0.5 (default) and mix=decay-mapped-something? NO — simpler: define
    // a NEW lambda body that maps `Second decay` to the underlying Reverb.Apply API.
    // Reverb.Apply takes (buffer, roomSize, damping, mix) — the existing 3-arg path.
    // For the new (Buffer, Double, Second) overload, the ergonomic semantic is:
    //   roomSize = arg1, decaySec = arg2; damping/mix derived (default damping=0.5, mix=0.3 like reverbSimple).
    var reverbSecondSig = new FunctionSignature("reverb",
        [BufferType.Instance, DoubleType.Instance, SecondType.Instance]);
    registry.Register("reverb", reverbSecondSig, args =>
    {
        var buffer = args[0].As<AudioBuffer>();
        float roomSize = (float)args[1].As<double>();
        float decaySec = (float)args[2].As<double>();   // Second's data IS a double
        // Derive damping from decay: longer decay => less damping (approximate inverse map).
        // Bound to reasonable ranges. This is a single new lambda body — no shared logic with ReverbFull.
        float damping = (float)Math.Clamp(0.7 - decaySec * 0.15, 0.1, 0.7);
        float mix = 0.3f;  // matches ReverbSimple default

        if (buffer.Frames == 0)
            return Value.Buffer(new AudioBuffer(0, buffer.Channels, buffer.SampleRate));

        var result = Reverb.Apply(buffer, roomSize, damping, mix);
        return Value.Buffer(result);
    });
}
```
