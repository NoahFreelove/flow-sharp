private Token ScanNumberOrSpecialLiteral(SourceLocation start)
{
    var sb = new StringBuilder();

    // Consume digits...
    while (!IsAtEnd() && char.IsDigit(Peek())) sb.Append(Advance());

    // Decimal point...
    if (!IsAtEnd() && Peek() == '.' && char.IsDigit(PeekNext()))
    {
        sb.Append(Advance());
        while (!IsAtEnd() && char.IsDigit(Peek())) sb.Append(Advance());
    }

    var numberText = sb.ToString();

    // Suffix dispatch — kHz BEFORE Hz, both BEFORE existing arms (or interleaved correctly):
    if (!IsAtEnd())
    {
        // NEW: Try "kHz" suffix FIRST (3 chars)
        if (Peek() == 'k' && PeekNext() == 'H' && _position + 2 < _source.Length && _source[_position + 2] == 'z')
        {
            Advance(); Advance(); Advance(); // consume k, H, z
            sb.Append('k'); sb.Append('H'); sb.Append('z');
            var text = sb.ToString();
            if (double.TryParse(numberText, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double kHzValue))
                return new Token(TokenType.HertzLiteral, text, start, kHzValue * 1000.0);  // canonical Hz
        }
        // NEW: Try "Hz" suffix (2 chars)
        else if (Peek() == 'H' && PeekNext() == 'z')
        {
            Advance(); Advance(); // consume H, z
            sb.Append('H'); sb.Append('z');
            var text = sb.ToString();
            if (double.TryParse(numberText, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double hzValue))
                return new Token(TokenType.HertzLiteral, text, start, hzValue);
        }
        // EXISTING: ms / dB / c / s arms below (unchanged) ...
        else if (Peek() == 'm' && PeekNext() == 's') { /* existing */ }
        else if (Peek() == 'd' && PeekNext() == 'B') { /* existing */ }
        else if (Peek() == 'c' && !char.IsLetter(PeekNext())) { /* existing */ }
        else if (Peek() == 's' && PeekNext() != 't') { /* existing */ }
    }

    // Fall through to int/float
}
