```csharp
/// <summary>
/// Phase 28: Articulation-aware ADSR envelope. Applies the LOCKED per-articulation
/// envelope shaping rules from SPEC-5 on top of the synthesizer's baseline ADSR.
///
/// Locked rules (SPEC-4 / SPEC-5):
///   Staccato + Marcato: attack × 0.66 (1.5× faster), sustain = 0, release × 0.5
///   Tenuto:             release × 1.2 (soft release)
///   Legato:             synth-default (crossfade overlap handled at BarRenderer via DurationOverlap)
///   Accent + Normal:    synth-default
///   Sforzando:          synth-default ADSR + time-varying spike (1.5× → 1.0× linear) over first 15% of frames
///
/// Drum synth passes isPercussion: true to bypass shaping (drums are inherently percussive).
/// </summary>
public static float[] GenerateArticulationADSR(
    Articulation articulation,
    double baseAttack, double baseDecay, double baseSustain, double baseRelease,
    int frames, int sampleRate, bool isPercussion = false)
{
    if (isPercussion)
        return GenerateADSR(baseAttack, baseDecay, baseSustain, baseRelease, frames, sampleRate);

    double attack = baseAttack;
    double decay = baseDecay;
    double sustain = baseSustain;
    double release = baseRelease;

    switch (articulation)
    {
        case Articulation.Staccato:
        case Articulation.Marcato:
            attack  = baseAttack * 0.66;  // 1.5× faster — sharper transient
            sustain = 0.0;                // zero sustain — pure attack-then-decay
            release = baseRelease * 0.5;  // fast release
            break;
        case Articulation.Tenuto:
            release = baseRelease * 1.2;  // soft release — held to full value
            break;
        case Articulation.Legato:
        case Articulation.Accent:
        case Articulation.Sforzando:
        case Articulation.Normal:
            // Synth-default ADSR. Sforzando spike applies post-curve below.
            break;
    }

    var curve = GenerateADSR(attack, decay, sustain, release, frames, sampleRate);

    if (articulation == Articulation.Sforzando)
    {
        int spikeFrames = (int)(frames * 0.15);
        for (int i = 0; i < spikeFrames; i++)
        {
            float t = (float)i / Math.Max(1, spikeFrames);
            // 1.5× at i=0, decays linearly to 1.0× at i=spikeFrames
            float multiplier = 1.5f * (1.0f - t) + 1.0f * t;
            curve[i] *= multiplier;
        }
    }

    return curve;
}
```

```csharp
public class PerSynthArticulationTests
{
    public static IEnumerable<object[]> SynthArticulationCombos =>
        new[] { "piano", "brass", "sax", "drums", "bell", "flute", "organ", "strings", "wavetable" }
            .SelectMany(synth => new[] { Articulation.Staccato, Articulation.Tenuto, Articulation.Marcato,
                                          Articulation.Accent, Articulation.Sforzando, Articulation.Legato }
                .Select(art => new object[] { synth, art }));

    [Theory]
    [MemberData(nameof(SynthArticulationCombos))]
    public void PerSynthArticulation_NormalVsArticulated_FFTCosineDifferentiable(string synthName, Articulation art)
    {
        // Render same note (C4q at BPM 120) under Normal and under `art` for `synthName`
        var normalBuffer = RenderC4q(synthName, Articulation.Normal);
        var articulatedBuffer = RenderC4q(synthName, art);

        // Compute FFT magnitude bins for both (length-aligned)
        float[] normalFft = ComputeFFTMagnitude(normalBuffer);
        float[] artFft = ComputeFFTMagnitude(articulatedBuffer);

        // Cosine similarity
        double cos = CosineSimilarity(normalFft, artFft);

        if (synthName == "drums")
        {
            // Drums are no-op — Normal and articulated should be ≥ 0.99 similar
            Assert.True(cos >= 0.99, $"Drums (no-op) expected high similarity, got {cos:F3} for {art}");
        }
        else
        {
            Assert.True(cos < 0.95, $"{synthName} {art} expected cosine < 0.95, got {cos:F3}");
        }
    }
}
```

Helpers:
- `RenderC4q(string synthName, Articulation art)` — uses BarRenderer.RenderBarToVoices with one C4q note and the chosen articulation; returns the resulting AudioBuffer.
- `ComputeFFTMagnitude(AudioBuffer)` — implement DFT (slow but tests are short, frames ≈ 0.5 sec × 44100 = 22050; bin set ≤ 256 for speed) OR a simple Goertzel-band approach computing magnitude at C4 frequency × harmonics. SIMPLER: just compute RMS in 10 frequency bands (0-200Hz, 200-400Hz, ..., 1800-2000Hz) using bandpass filtering with the existing Filter class. Treat the 10 RMS values as the FFT-bin proxy. This avoids implementing FFT.
- `CosineSimilarity(float[] a, float[] b)` — standard `dot(a,b) / (norm(a) × norm(b))`.

Test runtime budget: 54 facts × 0.5 sec render + ≤ 0.1 sec FFT = ~30 seconds total per SPEC constraints.</action>
