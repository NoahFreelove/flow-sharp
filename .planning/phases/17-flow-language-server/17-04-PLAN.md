**CRITICAL EXECUTOR NOTES:**
- **D-05 (standard scopes only)**: Use the LSP `SemanticTokenType` enum values (Keyword, Type, String, Number, Operator, Comment, Variable, Function, Macro) — these ARE standard. No custom legend names.
- **Tokens MUST be sorted by (line, column) ascending.** `SimpleLexer.Tokenize()` already guarantees this (verify via `flow-lang/Lexing/SimpleLexer.cs` — tokens produced in source order).
- **Skip unmapped tokens**, do not emit them with type index 0. Emitting zeros corrupts subsequent delta math (all following tokens' deltas would be wrong).
- **Same-line delta** uses `currCol - prevCol` (relative); **cross-line delta** uses `currCol` directly (absolute). Easy to flip — golden tests catch it.
- **0-based everywhere**: LSP Position is 0-based; Flow SourceLocation is 1-based. Subtract 1.

       ```csharp
       using System.Collections.Generic;
       using System.Threading;
       using System.Threading.Tasks;
       using FlowLang.Lexing;
       using OmniSharp.Extensions.LanguageServer.Protocol;
       using OmniSharp.Extensions.LanguageServer.Protocol.Client.Capabilities;
       using OmniSharp.Extensions.LanguageServer.Protocol.Document.Proposals;  // or .Document, depending on version
       using OmniSharp.Extensions.LanguageServer.Protocol.Models;
       using OmniSharp.Extensions.LanguageServer.Protocol.Models.Proposals;
       using OmniSharp.Extensions.LanguageServer.Protocol.Server.Capabilities;

       namespace FlowLsp.Handlers;

       public sealed class SemanticTokensHandler : SemanticTokensHandlerBase
       {
           private readonly DocumentManager _docs;
           private readonly ParseSession _parser;

           public SemanticTokensHandler(DocumentManager docs, ParseSession parser)
           {
               _docs = docs;
               _parser = parser;
           }

           // Legend — MUST mirror LegendIndex below.
           public static readonly SemanticTokenType[] Legend = new[] {
               SemanticTokenType.Keyword,    // 0
               SemanticTokenType.Type,       // 1
               SemanticTokenType.String,     // 2
               SemanticTokenType.Number,     // 3
               SemanticTokenType.Operator,   // 4
               SemanticTokenType.Comment,    // 5
               SemanticTokenType.Variable,   // 6 — note literals
               SemanticTokenType.Function,   // 7 — chord literals
               SemanticTokenType.Macro,      // 8 — pipe delimiters
           };

           public static readonly SemanticTokenModifier[] ModifierLegend =
               System.Array.Empty<SemanticTokenModifier>();

           private enum LegendIndex {
               Keyword = 0, Type = 1, String = 2, Number = 3, Operator = 4,
               Comment = 5, Variable = 6, Function = 7, Macro = 8
           }

           public static int? MapTokenType(TokenType t) => t switch {
               TokenType.Proc or TokenType.EndProc or TokenType.Return or
               TokenType.Use or TokenType.Internal or TokenType.Lazy or TokenType.Fn or
               TokenType.Section or TokenType.For or TokenType.While or TokenType.Break or
               TokenType.Continue or TokenType.In or TokenType.Progression or
               TokenType.Tempo or TokenType.Timesig or TokenType.Key or TokenType.Swing or
               TokenType.Dynamics or TokenType.Rit or TokenType.Accel or TokenType.Pickup or
               TokenType.Pan or TokenType.Gain
                   => (int)LegendIndex.Keyword,

               TokenType.Void or TokenType.Int or TokenType.Float or TokenType.Long or
               TokenType.Double or TokenType.String or TokenType.Bool or TokenType.Number or
               TokenType.Note or TokenType.Buf
                   => (int)LegendIndex.Type,

               TokenType.StringLiteral => (int)LegendIndex.String,
               TokenType.BoolLiteral => (int)LegendIndex.Number,  // Bool is a constant — LSP maps to Number bucket for now

               TokenType.IntLiteral or TokenType.FloatLiteral or
               TokenType.SemitoneLiteral or TokenType.CentLiteral or
               TokenType.TimeLiteral or TokenType.DecibelLiteral
                   => (int)LegendIndex.Number,

               TokenType.Arrow or TokenType.FatArrow or TokenType.Plus or
               TokenType.Minus or TokenType.Star or TokenType.Slash or
               TokenType.LessThan or TokenType.GreaterThan or TokenType.Assign
                   => (int)LegendIndex.Operator,

               TokenType.Comment => (int)LegendIndex.Comment,
               TokenType.NoteLiteral => (int)LegendIndex.Variable,
               TokenType.ChordLiteral => (int)LegendIndex.Function,
               TokenType.Pipe => (int)LegendIndex.Macro,

               _ => null
           };

           /// <summary>
           /// Encode Flow tokens into LSP's 5-tuple delta format.
           /// Per LSP 3.17 spec: [deltaLine, deltaStartChar, length, tokenType, tokenModifiers].
           /// Same-line deltas use column-offset; new-line deltas use absolute column.
           /// Tokens not in the legend are SKIPPED (not emitted as zeros — that corrupts subsequent deltas).
           /// </summary>
           public static int[] EncodeTokens(IReadOnlyList<Token> tokens)
           {
               var data = new List<int>(tokens.Count * 5);
               int prevLine = 0, prevCol = 0;
               bool first = true;
               foreach (var t in tokens)
               {
                   var typeIdx = MapTokenType(t.Type);
                   if (typeIdx is null) continue;
                   int line = t.Location.Line - 1;  // LSP 0-based
                   int col  = t.Location.Column - 1;
                   if (col < 0) col = 0;
                   if (line < 0) line = 0;
                   int dLine = first ? line : line - prevLine;
                   int dCol  = (first || dLine > 0) ? col : col - prevCol;
                   data.Add(dLine);
                   data.Add(dCol);
                   data.Add(t.Text?.Length ?? 0);
                   data.Add(typeIdx.Value);
                   data.Add(0);  // no modifiers for v1
                   prevLine = line;
                   prevCol = col;
                   first = false;
               }
               return data.ToArray();
           }

           // OmniSharp handler implementation — the exact method signatures depend on 0.19.9's
           // SemanticTokensHandlerBase shape. Consult the OmniSharp sample server for the override list.
           // Skeleton:
           protected override async Task Tokenize(SemanticTokensBuilder builder,
               ITextDocumentIdentifierParams identifier, CancellationToken cancellationToken)
           {
               var uri = identifier.TextDocument.Uri;
               var text = _docs.GetText(uri) ?? string.Empty;
               var result = _parser.Parse(text, uri.GetFileSystemPath());
               foreach (var t in result.Tokens)
               {
                   var idx = MapTokenType(t.Type);
                   if (idx is null) continue;
                   if (t.Text == null || t.Text.Length == 0) continue;
                   var line = System.Math.Max(0, t.Location.Line - 1);
                   var col  = System.Math.Max(0, t.Location.Column - 1);
                   builder.Push(line, col, t.Text.Length, Legend[idx.Value], null);
               }
               await Task.CompletedTask;
           }

           protected override Task<SemanticTokensDocument> GetSemanticTokensDocument(
               ITextDocumentIdentifierParams @params, CancellationToken cancellationToken)
               => Task.FromResult(new SemanticTokensDocument(RegistrationOptions));

           protected override SemanticTokensRegistrationOptions CreateRegistrationOptions(
               SemanticTokensCapability capability, ClientCapabilities clientCapabilities)
               => new()
               {
                   DocumentSelector = DocumentSelector.ForLanguage("flow"),
                   Legend = new OmniSharp.Extensions.LanguageServer.Protocol.Models.SemanticTokensLegend
                   {
                       TokenTypes = Legend,
                       TokenModifiers = ModifierLegend
                   },
                   Full = true,
                   Range = false
               };
       }
       ```

       ```csharp
       using System.Linq;
       using FlowLang.Lexing;
       using FlowLsp.Handlers;
       using OmniSharp.Extensions.LanguageServer.Protocol.Models;
       using Xunit;

       namespace FlowLang.Tests.Unit.Phase17;

       public class SemanticTokensTests
       {
           private static IReadOnlyList<Token> Tokens(string source) =>
               LspFixtures.Parse(source).Tokens;

           [Fact]
           public void EncodeTokens_EmptyBuffer_ReturnsEmpty()
           {
               // Only Eof token, which maps to null → skipped.
               var encoded = SemanticTokensHandler.EncodeTokens(Tokens(""));
               Assert.Empty(encoded);
           }

           [Fact]
           public void EncodeTokens_SingleKeyword_Emits5Tuple()
           {
               var tokens = Tokens("proc");
               var proc = tokens.First(t => t.Type == TokenType.Proc);
               var encoded = SemanticTokensHandler.EncodeTokens(new[] { proc });
               Assert.Equal(5, encoded.Length);
               Assert.Equal(0, encoded[0]); // deltaLine (first token, absolute)
               Assert.Equal(proc.Location.Column - 1, encoded[1]); // deltaStartChar
               Assert.Equal(4, encoded[2]); // "proc".Length
               var keywordIdx = System.Array.IndexOf(SemanticTokensHandler.Legend, SemanticTokenType.Keyword);
               Assert.Equal(keywordIdx, encoded[3]);
               Assert.Equal(0, encoded[4]); // no modifiers
           }

           [Fact]
           public void EncodeTokens_SameLineDelta_UsesColumnOffset()
           {
               // Two keywords on the same line: "proc main" — note `main` is Identifier (unmapped).
               // Use two keywords instead: e.g. "proc return" is syntactically bogus but lexes cleanly.
               var tokens = Tokens("proc return").Where(t => t.Type != TokenType.Eof).ToList();
               Assert.True(tokens.Count >= 2);
               // Call EncodeTokens on only Proc + Return tokens to force same-line delta.
               var procTok = tokens.First(t => t.Type == TokenType.Proc);
               var retTok  = tokens.First(t => t.Type == TokenType.Return);
               var encoded = SemanticTokensHandler.EncodeTokens(new[] { procTok, retTok });
               Assert.Equal(10, encoded.Length); // 2 tokens × 5 ints
               // Second token same line → dLine=0, dCol = retCol - procCol
               Assert.Equal(0, encoded[5]);
               Assert.Equal(retTok.Location.Column - procTok.Location.Column, encoded[6]);
           }

           [Fact]
           public void EncodeTokens_CrossLineDelta_UsesAbsoluteColumn()
           {
               var tokens = Tokens("proc\n  return").Where(t => t.Type != TokenType.Eof).ToList();
               var procTok = tokens.First(t => t.Type == TokenType.Proc);
               var retTok  = tokens.First(t => t.Type == TokenType.Return);
               var encoded = SemanticTokensHandler.EncodeTokens(new[] { procTok, retTok });
               // dLine = 1, dCol = absolute column of return (2 in 0-based since "  return")
               Assert.Equal(1, encoded[5]);
               Assert.Equal(retTok.Location.Column - 1, encoded[6]);
           }

           [Fact]
           public void EncodeTokens_SkipsUnmappedTokens()
           {
               // Identifier alone (no keyword match) produces no output.
               // "xyz" — if SimpleLexer classifies it as Identifier (not in Legend), result is empty.
               var tokens = Tokens("xyz");
               var encoded = SemanticTokensHandler.EncodeTokens(tokens);
               Assert.Empty(encoded);
           }

           [Fact]
           public void MapTokenType_Keyword_HasIndex() =>
               Assert.NotNull(SemanticTokensHandler.MapTokenType(TokenType.Proc));

           [Fact]
           public void MapTokenType_NoteLiteral_MapsToVariable()
           {
               var idx = SemanticTokensHandler.MapTokenType(TokenType.NoteLiteral);
               Assert.NotNull(idx);
               Assert.Equal(SemanticTokenType.Variable, SemanticTokensHandler.Legend[idx.Value]);
           }

           [Fact]
           public void MapTokenType_ChordLiteral_MapsToFunction()
           {
               var idx = SemanticTokensHandler.MapTokenType(TokenType.ChordLiteral);
               Assert.NotNull(idx);
               Assert.Equal(SemanticTokenType.Function, SemanticTokensHandler.Legend[idx.Value]);
           }

           [Fact]
           public void MapTokenType_UnknownToken_ReturnsNull() =>
               Assert.Null(SemanticTokensHandler.MapTokenType(TokenType.Eof));
       }
       ```
