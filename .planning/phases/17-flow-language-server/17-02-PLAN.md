**CRITICAL EXECUTOR NOTES:**
- **Standard scopes ONLY** per D-05. Never invent `*.flow`-specific scopes (e.g., `keyword.music.tempo.flow`). Keyword-vs-music-keyword discrimination is the semantic-tokens handler's job (plan 17-04), not the TM grammar's. Pitfall: invented scopes render unstyled in user themes.
- **Comment syntax is `//` ONLY.** Confirmed by reading `flow-lang/Lexing/SimpleLexer.cs:826-833` — the lexer recognizes `//` for line comments and the CLAUDE.md-documented `Note:` line-start marker; it does NOT recognize `;`. Do NOT add a `;` fallback branch to the TM grammar — it would color semicolons (valid statement terminators in Flow) as comments. Language-configuration.json commits to `//` as the one and only `lineComment`.
- **Do NOT model bar-boundaries** with `begin`/`end` in note streams — TM grammars are line-oriented and get note-stream boundaries wrong (RESEARCH Pitfall 5). Color notes/chords/durations as standalone patterns; semantic tokens later refine.
- **vsce target names ≠ .NET RIDs**: platformDir() must produce `linux-x64`, `win32-x64` (NOT `win-x64`), `darwin-x64`, `darwin-arm64` (Pitfall 7). process.platform returns `win32`, NOT `win`.
- **.gitignore must exclude** `vscode-extension/node_modules/`, `vscode-extension/out/`, `vscode-extension/server/` (populated at build time by CI), `*.vsix`. Do NOT commit node_modules.

       ```typescript
       import * as path from 'path';
       import * as fs from 'fs';
       import { workspace, ExtensionContext, window } from 'vscode';
       import { LanguageClient, LanguageClientOptions, ServerOptions, Executable, TransportKind } from 'vscode-languageclient/node';

       let client: LanguageClient | undefined;

       function platformDir(): string {
         const platform = process.platform;  // 'linux' | 'win32' | 'darwin'
         const arch = process.arch;          // 'x64' | 'arm64'
         return `${platform}-${arch}`;       // e.g. 'linux-x64', 'darwin-arm64'
       }

       function defaultBinaryPath(context: ExtensionContext): string {
         const dir = platformDir();
         const exe = process.platform === 'win32' ? 'flow-lsp.exe' : 'flow-lsp';
         return context.asAbsolutePath(path.join('server', dir, exe));
       }

       export async function activate(context: ExtensionContext) {
         const config = workspace.getConfiguration('flow');
         const override = (config.get<string>('server.path') ?? '').trim();
         const binary = override !== '' ? override : defaultBinaryPath(context);

         if (!fs.existsSync(binary)) {
           window.showErrorMessage(`Flow LSP binary not found at ${binary}`);
           return;
         }

         if (process.platform !== 'win32') {
           try { fs.chmodSync(binary, 0o755); } catch { /* best-effort */ }
         }

         const exe: Executable = {
           command: binary,
           transport: TransportKind.stdio,
           options: { env: process.env }
         };
         const serverOptions: ServerOptions = { run: exe, debug: exe };

         const clientOptions: LanguageClientOptions = {
           documentSelector: [{ scheme: 'file', language: 'flow' }],
           synchronize: {
             fileEvents: workspace.createFileSystemWatcher('**/*.flow')
           },
           traceOutputChannel: window.createOutputChannel('Flow LSP Trace')
         };

         client = new LanguageClient('flow', 'Flow Language Server', serverOptions, clientOptions);
         await client.start();
       }

       export function deactivate(): Thenable<void> | undefined {
         return client?.stop();
       }
       ```
