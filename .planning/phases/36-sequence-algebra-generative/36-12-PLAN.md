// =====================================================================
// Markov Jazz — Phase 36 generative showcase
// =====================================================================
// Demonstrates:
//   - one-shot (markov corpus order length seed)
//   - markovTrain → MarkovModel → markovGenerate split (D-36-06)
//   - (jam over=chords style=#jazz key=... seed=...) chord-aware improv
//   - Phase 35 `as` chain naming on Phase 36 combinators
//
// Cross-platform note: Lorenz / chaos primitives have platform-specific
// FP divergence per D-36-09; this example uses Markov + jam only.
//
// Render with:
//   flow render examples/generative/markov_jazz.flow

use "@std"
use "@audio"
use "@patterns"
use "@generative"
use "@improv"

voicePool 32 {
    tempo 120 {
        key Cmajor {
            // Train once, generate many
            Sequence corpus = | C4 D4 E4 F4 G4 F4 E4 D4 C4 |
            MarkovModel m = (markovTrain corpus 2)
            Sequence riff = (markovGenerate m 16 42)

            // Chord-aware improv
            Sequence chords = | Cmaj7 Am7 Dm7 G7 |
            Sequence solo = (jam over=chords style=#jazz length=8 seed=1234)
                -> (every 4 (fn s => (fast s 2))) as varied
                -> render

            // Render to WAV
            Song song = [solo*2]
            (writeWav "examples/output/markov_jazz.wav" (renderSong song))
        }
    }
}
```

use "@std"
use "@audio"

// Phase 36 parameterized sections — D-36-13..18
tempo 120 {
    key Cmajor {

        // Overload 1 — single Note pattern, specificity 500
        section verse(Note root) {
            | $root q E4 q G4 q B4 q |
        }

        // Overload 2 — tuple destructure, specificity 600
        section verse(<<Note root, Int repeats>>) {
            // emit one bar per repeat
            for i in (range 0 repeats) { | $root q E4 q G4 q B4 q | }
        }

        // Overload 3 — constructor pattern w/ music-aware extractor, specificity 800
        section verse(Cmaj7) {
            | C4 q E4 q G4 q B4 q | C5 q B4 q G4 q E4 q |
        }

        section chorus {
            | C4 h E4 h G4 h B4 h |
        }

        Song song = [
            verse(C4)             // Overload 1
            verse(<<D4, 3>>)      // Overload 2
            verse(Cmaj7)          // Overload 3 (800 wins)
            chorus
            verse(C4)*3           // Overload 1 repeated 3x via D-36-14
        ]

        (writeWav "examples/output/parameterized.wav" (renderSong song))
    }
}
```
