Create flow-cli/Config/FlowConfigLoader.cs:
```
using FlowLang.Runtime;
using Tomlyn;

namespace FlowCli.Config;

public static class FlowConfigLoader
{
    // Path: ~/.config/flow/config.toml (XDG convention; per SPEC-4 / RESEARCH Open Question 5)
    // Missing file: silent fallback to FlowConfigPoco.Defaults (no warning, no error).
    // Malformed file: single 'Warning:' line to stderr; fall back to Defaults; DO NOT abort.
    public static void LoadFromXdg()
    {
        var path = Path.Combine(
            Environment.GetFolderPath(Environment.SpecialFolder.UserProfile),
            ".config", "flow", "config.toml");

        if (!File.Exists(path))
        {
            // Silent fallback per SPEC-4 acceptance.
            return;
        }

        try
        {
            var text = File.ReadAllText(path);
            FlowConfig.Active = Toml.ToModel<FlowConfigPoco>(text);
        }
        catch (Exception ex) when (ex is TomlException || ex is IOException)
        {
            // Charitable: warn but do not abort (per CLAUDE.md feedback_charitable_interpretation memory).
            Console.Error.WriteLine($"Warning: {path} could not be parsed: {ex.Message}");
            Console.Error.WriteLine("Falling back to baked-in defaults.");
            FlowConfig.Active = FlowConfigPoco.Defaults;
        }
    }
}
```
