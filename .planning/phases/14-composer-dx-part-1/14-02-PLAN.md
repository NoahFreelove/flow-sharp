    ```csharp
    private static Value Enharmonic(IReadOnlyList<Value> args, FlowLang.Runtime.ExecutionContext context)
    {
        // Note values are stored as their string form in the Value wrapper (Value.cs:32 region).
        string noteStr = args[0].As<string>();
        var (letter, octave, alteration) = NoteType.Parse(noteStr);

        var musicalCtx = context.GetMusicalContext();
        string? key = musicalCtx?.Key;

        // No-key / Cmaj / Amin fallback (D-05) — naturals always return unchanged.
        if (alteration == 0)
            return Value.Note(NoteType.Format(letter, octave, 0));

        // Compute MIDI of input. We will respell against this MIDI value.
        int inputMidi = NoteType.ToMidiNote(letter, octave, alteration);

        // In-key branch (D-04): if input pitch is diatonic to the active key, return scale-diatonic spelling.
        if (key != null)
        {
            // Per RESEARCH Pitfall 3: ScaleDatabase.GetScaleNotes returns sharp-spelled notes
            // even for flat keys (ChromaticNotes is sharp-only). Do NOT echo its string output.
            // Instead: enumerate the scale's MIDI values at the input octave; if input MIDI matches
            // any scale-tone MIDI, infer the spelling whose key affinity (flat key → flat letter,
            // sharp key → sharp letter) matches.

            var scaleNotes = ScaleDatabase.GetScaleNotes(key);
            if (scaleNotes != null)
            {
                bool isFlatKey = key.Contains('b') || key.StartsWith("F", StringComparison.OrdinalIgnoreCase);
                // Iterate scale tones, compute MIDI for each at the input's octave region,
                // find a match for inputMidi % 12, infer the canonical spelling.
                int targetSemitone = ((inputMidi % 12) + 12) % 12;
                foreach (var scaleStr in scaleNotes)
                {
                    // scaleStr could be "C", "Db", "Cs", "D", "Eb", "Ds", etc.
                    // Convert to (letter, alt) pair and compute its semitone using NoteToSemitone via ScaleDatabase.
                    // If match: synthesize spelling preferring flats for flat keys.
                    if (TryGetSemitoneOfScaleSpelling(scaleStr, out int sem) && sem == targetSemitone)
                    {
                        var (sLetter, sAlt) = ParseScaleNoteToken(scaleStr, isFlatKey);
                        // Octave aligned to inputMidi's actual octave bucket
                        int octBucket = inputMidi / 12 - 1;
                        // Recompute octave so that ToMidiNote(sLetter, oct, sAlt) == inputMidi
                        int oct = octBucket;
                        // Calibrate oct so result matches inputMidi
                        while (NoteType.ToMidiNote(sLetter, oct, sAlt) > inputMidi) oct--;
                        while (NoteType.ToMidiNote(sLetter, oct, sAlt) < inputMidi) oct++;
                        return Value.Note(NoteType.Format(sLetter, oct, sAlt));
                    }
                }
                // chromatic-not-in-scale → fall through to no-key flip
            }
        }

        // No-key flip (D-05): sharp ↔ flat. Naturals were already returned above.
        // sharp → choose adjacent letter UP whose natural MIDI minus inputMidi gives negative alt.
        // flat  → choose adjacent letter DOWN whose natural MIDI minus inputMidi gives positive alt.
        // For double-sharps/flats, the result may be a natural (e.g., F##4 → G4) — D-05 documents
        // non-involutive cases.
        var (flippedLetter, flippedOct, flippedAlt) = ComputeFlippedSpelling(letter, octave, alteration, inputMidi);
        return Value.Note(NoteType.Format(flippedLetter, flippedOct, flippedAlt));
    }

    private static bool TryGetSemitoneOfScaleSpelling(string token, out int semitone)
    {
        // ScaleDatabase.NoteToSemitone covers C, D, E, F, G, A, B and Csharp, Cs, Db, Ds, Eb, ...
        // Use its public lookup if exposed; else replicate inline.
        // Implementer: read ScaleDatabase.cs and use its existing NoteToSemitone dictionary
        // accessor (or expose a helper if private).
        semitone = 0;
        try
        {
            // Fallback: parse token directly.
            char l = char.ToUpper(token[0]);
            int baseSem = l switch
            {
                'C' => 0, 'D' => 2, 'E' => 4, 'F' => 5, 'G' => 7, 'A' => 9, 'B' => 11,
                _ => -1
            };
            if (baseSem < 0) return false;
            int delta = 0;
            for (int i = 1; i < token.Length; i++)
            {
                switch (token[i])
                {
                    case 's': case '#': case '+': delta++; break;
                    case 'b': case '-': delta--; break;
                    default:
                        // unexpected char in scale spelling — give up
                        return false;
                }
            }
            semitone = ((baseSem + delta) % 12 + 12) % 12;
            return true;
        }
        catch
        {
            return false;
        }
    }

    private static (char letter, int alt) ParseScaleNoteToken(string token, bool preferFlat)
    {
        char letter = char.ToUpper(token[0]);
        int alt = 0;
        for (int i = 1; i < token.Length; i++)
        {
            switch (token[i])
            {
                case 's': case '#': case '+': alt++; break;
                case 'b': case '-': alt--; break;
            }
        }
        // If the scale token came back sharp-spelled (e.g., "Cs" for the same MIDI as "Db"),
        // and the key prefers flats, rewrite to flat-letter neighbor.
        if (preferFlat && alt > 0)
        {
            // Convert sharp to flat: letter UP, alt = -1
            char up = letter switch
            {
                'C' => 'D', 'D' => 'E', 'E' => 'F', 'F' => 'G',
                'G' => 'A', 'A' => 'B', 'B' => 'C',
                _ => letter
            };
            return (up, alt - (NaturalSemitoneOf(up) - NaturalSemitoneOf(letter)));
            // For 'Cs' (sharp 1, MIDI semitone 1): up='D' (semitone 2), alt = 1 - (2 - 0) = -1 → 'D' alt -1 = 'Db'.
        }
        return (letter, alt);
    }

    private static int NaturalSemitoneOf(char letter) => letter switch
    {
        'C' => 0, 'D' => 2, 'E' => 4, 'F' => 5, 'G' => 7, 'A' => 9, 'B' => 11, _ => 0
    };

    private static (char letter, int oct, int alt) ComputeFlippedSpelling(char letter, int octave, int alteration, int inputMidi)
    {
        if (alteration > 0)
        {
            // sharp → letter UP, alt = inputMidi - naturalMidi(up)
            char up = letter switch
            {
                'C' => 'D', 'D' => 'E', 'E' => 'F', 'F' => 'G',
                'G' => 'A', 'A' => 'B', 'B' => 'C',
                _ => letter
            };
            int upOct = (letter == 'B') ? octave + 1 : octave;
            int upMidi = NoteType.ToMidiNote(up, upOct, 0);
            int newAlt = inputMidi - upMidi;
            return (up, upOct, newAlt);
        }
        else
        {
            // flat → letter DOWN, alt = inputMidi - naturalMidi(down)
            char down = letter switch
            {
                'C' => 'B', 'D' => 'C', 'E' => 'D', 'F' => 'E',
                'G' => 'F', 'A' => 'G', 'B' => 'A',
                _ => letter
            };
            int downOct = (letter == 'C') ? octave - 1 : octave;
            int downMidi = NoteType.ToMidiNote(down, downOct, 0);
            int newAlt = inputMidi - downMidi;
            return (down, downOct, newAlt);
        }
    }
    ```
